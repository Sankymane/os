#!/bin/bash
echo "Enter first number:"
read num1

echo "Enter second number:"
read num2

echo "Choose operation (+, -, *, /):"
read op

case $op in
  +)
    result=$((num1 + num2))
    echo "Result: $num1 + $num2 = $result"
    ;;
  -)
    result=$((num1 - num2))
    echo "Result: $num1 - $num2 = $result"
    ;;
  \*)
    result=$((num1 * num2))
    echo "Result: $num1 * $num2 = $result"
    ;;
  /)
    if [ $num2 -ne 0 ]; then
      result=$((num1 / num2))
      echo "Result: $num1 / $num2 = $result"
    else
      echo "Error: Division by zero is not allowed."
    fi
    ;;
  *)
    echo "Invalid operation selected."
    ;;
esac
