
public class FIFO {
    public static void main(String[] args) {

        int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3,
                       0, 3, 2, 1, 2, 0, 1, 7, 0, 1};

        int frames[] = new int[3];
        int index = 0, faults = 0, hits = 0;

        for (int i = 0; i < 3; i++)
            frames[i] = -1;

        for (int i = 0; i < pages.length; i++) {

            int found = 0;

            for (int j = 0; j < 3; j++) {
                if (frames[j] == pages[i]) {
                    found = 1;
                    hits++;
                    break;
                }
            }

            if (found == 0) {
                frames[index] = pages[i];
                index = (index + 1) % 3;
                faults++;
            }
        }

        System.out.println("FIFO Page Faults: " + faults);
        System.out.println("FIFO Page Hits: " + hits);
    }
}
