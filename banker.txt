
public class Bankers {
    public static void main(String[] args) {

        int n = 5;

        int alloc[][] = {
            {0, 1, 0},
            {2, 0, 0},
            {3, 0, 2},
            {2, 1, 1},
            {0, 0, 2}
        };

        int max[][] = {
            {7, 5, 3},
            {3, 2, 2},
            {9, 0, 2},
            {2, 2, 2},
            {4, 3, 3}
        };

        int avail[] = {3, 3, 2};

        int need[][] = new int[n][3];
        int finish[] = new int[n];

        // Calculate Need
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                need[i][j] = max[i][j] - alloc[i][j];
            }
        }

        int work[] = {3, 3, 2};
        int count = 0;

        System.out.println("Safe Sequence:");

        while (count < n) {
            int found = 0;

            for (int i = 0; i < n; i++) {

                if (finish[i] == 0) {

                    if (need[i][0] <= work[0] &&
                        need[i][1] <= work[1] &&
                        need[i][2] <= work[2]) {

                        work[0] += alloc[i][0];
                        work[1] += alloc[i][1];
                        work[2] += alloc[i][2];

                        finish[i] = 1;
                        count++;

                        System.out.print("P" + i + " ");
                        found = 1;
                    }
                }
            }

            if (found == 0) {
                break;
            }
        }

        if (count == n) {
            System.out.println("\nSystem is in SAFE state.");
        } else {
            System.out.println("\nSystem is in UNSAFE state.");
        }
    }
}
