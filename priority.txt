
public class Priority {
    public static void main(String[] args) {

        int bt[] = {5, 2, 3};
        int p[] = {1, 2, 3};
        int pr[] = {3, 1, 2};
        int wt[] = new int[3];
        int tat[] = new int[3];
        int temp;

        for (int i = 0; i < 3; i++) {
            for (int j = i+1; j < 3; j++) {

                if (pr[i] > pr[j]) {

                    temp = pr[i];
                    pr[i] = pr[j];
                    pr[j] = temp;

                    temp = bt[i];
                    bt[i] = bt[j];
                    bt[j] = temp;

                    temp = p[i];
                    p[i] = p[j];
                    p[j] = temp;
                }
            }
        }

        wt[0] = 0;

        for (int i = 1; i < 3; i++) {
            wt[i] = wt[i-1] + bt[i-1];
        }

        for (int i = 0; i < 3; i++) {
            tat[i] = wt[i] + bt[i];

            System.out.println("P" + p[i] +
                " WT=" + wt[i] + " TAT=" + tat[i]);
        }
    }
}
